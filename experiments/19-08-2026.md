# ============================================================
# DUPLICATE BEHAVIORAL VECTOR + TRAIN/VAL/TEST PREPARATION
# ============================================================

import pandas as pd
import numpy as np

from sklearn.model_selection import train_test_split

DATA_PATH = "/kaggle/working/Compact_API_DLL_Dataset.csv"

print("=" * 70)
print("DUPLICATE VECTOR ANALYSIS + DATA SPLIT")
print("=" * 70)

# ============================================================
# LOAD DATA
# ============================================================

df = pd.read_csv(DATA_PATH)

behavioral_cols = [
    c for c in df.columns
    if c not in ["SHA256", "Type"]
]

print(f"\nDataset shape: {df.shape}")
print(f"Behavioral features: {len(behavioral_cols)}")

# ============================================================
# STEP 1: DUPLICATE BEHAVIORAL VECTOR ANALYSIS
# ============================================================

print("\n" + "=" * 70)
print("STEP 1: DUPLICATE BEHAVIORAL VECTOR ANALYSIS")
print("=" * 70)

# Number of unique behavioral vectors
unique_vectors = df[behavioral_cols].drop_duplicates()

print(f"Total samples: {len(df):,}")
print(f"Unique behavioral vectors: {len(unique_vectors):,}")
print(
    f"Duplicate samples: "
    f"{len(df) - len(unique_vectors):,}"
)

# ------------------------------------------------------------
# Find behavioral vectors with multiple labels
# ------------------------------------------------------------

vector_label_counts = (
    df.groupby(behavioral_cols, sort=False)["Type"]
      .nunique()
)

conflicting_vectors = vector_label_counts[
    vector_label_counts > 1
]

print(
    f"\nBehavioral vectors appearing with multiple labels: "
    f"{len(conflicting_vectors):,}"
)

if len(conflicting_vectors) == 0:
    print(
        "PASS: Every identical behavioral vector has "
        "the same target label."
    )
else:
    print(
        "WARNING: Some identical behavioral vectors "
        "have conflicting labels."
    )

# ============================================================
# STEP 2: DUPLICATE VECTOR LABEL CONSISTENCY
# ============================================================

print("\n" + "=" * 70)
print("STEP 2: DUPLICATE VECTOR LABEL CONSISTENCY")
print("=" * 70)

vector_stats = (
    df.groupby(behavioral_cols, sort=False)["Type"]
      .agg(["count", "nunique"])
      .reset_index()
)

pure_duplicate_groups = vector_stats[
    vector_stats["count"] > 1
]

print(
    f"Repeated behavioral groups: "
    f"{len(pure_duplicate_groups):,}"
)

if len(pure_duplicate_groups) > 0:

    print("\nTop repeated behavioral vectors:")

    print(
        pure_duplicate_groups
        .sort_values("count", ascending=False)
        [["count", "nunique"]]
        .head(10)
        .to_string(index=False)
    )

# ============================================================
# STEP 3: CLASS DISTRIBUTION
# ============================================================

print("\n" + "=" * 70)
print("STEP 3: ORIGINAL CLASS DISTRIBUTION")
print("=" * 70)

print(
    df["Type"]
    .value_counts()
    .sort_index()
)

# ============================================================
# STEP 4: TRAIN / VALIDATION / TEST SPLIT
# ============================================================

print("\n" + "=" * 70)
print("STEP 4: STRATIFIED DATA SPLIT")
print("=" * 70)

# ------------------------------------------------------------
# First split:
# 80% train
# 20% temporary
# ------------------------------------------------------------

train_df, temp_df = train_test_split(
    df,
    test_size=0.20,
    stratify=df["Type"],
    random_state=42
)

# ------------------------------------------------------------
# Second split:
# temp -> 10% validation + 10% test
# ------------------------------------------------------------

val_df, test_df = train_test_split(
    temp_df,
    test_size=0.50,
    stratify=temp_df["Type"],
    random_state=42
)

print(f"Train      : {len(train_df):,}")
print(f"Validation : {len(val_df):,}")
print(f"Test       : {len(test_df):,}")

# ============================================================
# STEP 5: VERIFY SPLIT DISTRIBUTIONS
# ============================================================

print("\n" + "=" * 70)
print("STEP 5: CLASS DISTRIBUTION BY SPLIT")
print("=" * 70)

for name, subset in [
    ("TRAIN", train_df),
    ("VALIDATION", val_df),
    ("TEST", test_df)
]:

    print(f"\n{name}")
    print("-" * 40)

    counts = subset["Type"].value_counts().sort_index()

    percentages = (
        counts / len(subset) * 100
    ).round(2)

    summary = pd.DataFrame({
        "Samples": counts,
        "Percentage": percentages
    })

    print(summary)

# ============================================================
# STEP 6: CHECK SHA256 OVERLAP
# ============================================================

print("\n" + "=" * 70)
print("STEP 6: SHA256 OVERLAP CHECK")
print("=" * 70)

train_hashes = set(train_df["SHA256"])
val_hashes = set(val_df["SHA256"])
test_hashes = set(test_df["SHA256"])

train_val_overlap = len(train_hashes & val_hashes)
train_test_overlap = len(train_hashes & test_hashes)
val_test_overlap = len(val_hashes & test_hashes)

print(f"Train ∩ Validation : {train_val_overlap}")
print(f"Train ∩ Test       : {train_test_overlap}")
print(f"Validation ∩ Test  : {val_test_overlap}")

if (
    train_val_overlap == 0
    and train_test_overlap == 0
    and val_test_overlap == 0
):
    print("\nPASS: No SHA256 leakage between splits.")
else:
    print("\nWARNING: SHA256 overlap detected.")

# ============================================================
# STEP 7: CHECK BEHAVIORAL VECTOR OVERLAP
# ============================================================

print("\n" + "=" * 70)
print("STEP 7: BEHAVIORAL VECTOR OVERLAP")
print("=" * 70)

train_vectors = set(
    map(
        tuple,
        train_df[behavioral_cols].values
    )
)

val_vectors = set(
    map(
        tuple,
        val_df[behavioral_cols].values
    )
)

test_vectors = set(
    map(
        tuple,
        test_df[behavioral_cols].values
    )
)

print(
    f"Train unique vectors: "
    f"{len(train_vectors):,}"
)

print(
    f"Validation unique vectors: "
    f"{len(val_vectors):,}"
)

print(
    f"Test unique vectors: "
    f"{len(test_vectors):,}"
)

print(
    f"\nTrain ∩ Validation behavioral vectors: "
    f"{len(train_vectors & val_vectors):,}"
)

print(
    f"Train ∩ Test behavioral vectors: "
    f"{len(train_vectors & test_vectors):,}"
)

print(
    f"Validation ∩ Test behavioral vectors: "
    f"{len(val_vectors & test_vectors):,}"
)

# ============================================================
# STEP 8: SAVE SPLITS
# ============================================================

print("\n" + "=" * 70)
print("STEP 8: SAVING SPLITS")
print("=" * 70)

TRAIN_PATH = "/kaggle/working/Train_API_DLL.csv"
VAL_PATH = "/kaggle/working/Validation_API_DLL.csv"
TEST_PATH = "/kaggle/working/Test_API_DLL.csv"

train_df.to_csv(
    TRAIN_PATH,
    index=False
)

val_df.to_csv(
    VAL_PATH,
    index=False
)

test_df.to_csv(
    TEST_PATH,
    index=False
)

print(f"Train saved      : {TRAIN_PATH}")
print(f"Validation saved : {VAL_PATH}")
print(f"Test saved       : {TEST_PATH}")

# ============================================================
# FINAL SUMMARY
# ============================================================

print("\n" + "=" * 70)
print("DATA SPLIT COMPLETE")
print("=" * 70)

print(f"Train      : {len(train_df):,}")
print(f"Validation : {len(val_df):,}")
print(f"Test       : {len(test_df):,}")
print(f"Features   : {len(behavioral_cols):,}")
print("Random state: 42")
print("Split: 80 / 10 / 10")
